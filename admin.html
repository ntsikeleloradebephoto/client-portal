async function saveStockEdit(stockId) { if (!currentUser) return openLoginModal(); const sku = document.getElementById(`sku_${stockId}`).value.trim(); const itemName = document.getElementById(`name_${stockId}`).value.trim(); const supplier = document.getElementById(`supplier_${stockId}`).value.trim(); const qty = parseInt(document.getElementById(`qty_${stockId}`).value) || 0; const price = parseFloat(document.getElementById(`price_${stockId}`).value) || 0; const retailPrice = parseFloat(document.getElementById(`selling_${stockId}`).value) || 0; const { error } = await supabaseClient.from('inventory').update({ sku: sku, item_name: itemName, item_description: itemName, supplier: supplier, quantity_allocated: qty, quantity: qty, unit_price: price, cost_price: price, wholesale_floor_zar: price, retail_price: retailPrice // Matches your Supabase schema column exactly }).eq('id', stockId); if (error) alert('Failed to update stock: ' + error.message); else { logAuditAction(`Updated Stock Item [${sku}] ${itemName} by ${currentUser.full_name}`); alert('Stock item updated successfully!'); fetchAdminOrders(); } }
